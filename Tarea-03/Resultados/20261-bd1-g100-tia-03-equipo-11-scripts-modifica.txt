--
-- Modificación de Tablas
--

--
-- 1.1.- Agregar un campo a la tabla de usuarios de la red social
--

-- Se agrega el campo 'telefono' para que los usuarios puedan registrar
-- un número de contacto opcional en su perfil.
ALTER TABLE usuario
    ADD COLUMN IF NOT EXISTS telefono VARCHAR(20) NULL;


--
-- 1.2.- Modificar un campo de la tabla de usuarios
--

-- Se amplía el tamaño del campo 'nombre' de VARCHAR(100) a VARCHAR(150)
-- para soportar nombres compuestos más largos.
ALTER TABLE usuario
    ALTER COLUMN nombre TYPE VARCHAR(150);


--
-- Gestionar una tabla "nueva"
-- 1.- "agregar" una nueva tabla a la base de datos que tenga relación con el sistema
-- 2.- Darle un nombre "coherente"
-- 3.- Agregar campos coherentes con la tabla
-- 4.- Realizar todas las operaciones que se solicitan a continuación
--


-- 1.3.1
-- Crear una tabla "nueva" de su iniciativa (una tabla coherente con el sistema con su nombre, no coloque "nueva" como nombre)
--

-- Se crea la tabla 'insignia', que representa los logros o badges que
-- el sistema puede otorgar a los usuarios por su participación activa
-- en la plataforma (ej: "Primer Post", "10 Seguidores", "Mentor Activo").
CREATE TABLE insignia (
    id_insignia     SERIAL          NOT NULL,
    nombre          VARCHAR(100)    NOT NULL,
    descripcion     TEXT            NULL,
    icono_url       VARCHAR(255)    NULL
);


-- 1.3.2
-- Agregar una clave primaria y otros 3 campos cualquiera a la tabla "nueva"
-- Mínimo un campo tipo texto y uno numérico
--

-- Se define la clave primaria sobre id_insignia.
ALTER TABLE insignia
    ADD CONSTRAINT pk_insignia PRIMARY KEY (id_insignia);

-- Campo texto: categoría de la insignia (participación, liderazgo, conocimiento, etc.)
ALTER TABLE insignia
    ADD COLUMN IF NOT EXISTS categoria VARCHAR(80) NULL;

-- Campo numérico: puntos que otorga la insignia al usuario
ALTER TABLE insignia
    ADD COLUMN IF NOT EXISTS puntos INT NULL DEFAULT 0;

-- Campo texto: nivel de rareza de la insignia
ALTER TABLE insignia
    ADD COLUMN IF NOT EXISTS rareza VARCHAR(20) NULL DEFAULT 'comun';


-- 1.3.3
-- Quitar uno de los campos de la tabla "nueva"
--

-- Se elimina el campo 'icono_url' porque se decidió unificar los recursos
-- visuales a través del sistema de multimedia centralizado de la plataforma.
ALTER TABLE insignia
    DROP COLUMN IF EXISTS icono_url;


-- 1.3.4
-- Cambiar el nombre de la tabla "nueva" a otro nombre "otro_nombre"
-- Todas las operaciones siguientes se realizan sobre la tabla renombrada
--

-- Se renombra 'insignia' a 'logro' para reflejar mejor el concepto
-- de logro alcanzado por el usuario dentro de la red social.
ALTER TABLE insignia
    RENAME TO logro;


-- 1.3.5
-- Agregar un campo único a la tabla
--

-- Se agrega restricción UNIQUE al campo 'nombre' para garantizar que
-- no existan dos logros con el mismo nombre en la plataforma.
ALTER TABLE logro
    ADD CONSTRAINT uq_logro_nombre UNIQUE (nombre);


-- 1.3.6
-- Agregar 2 fechas de inicio y fin; y colocar un control de orden de fechas
--

-- Se agregan las fechas de vigencia del logro (campañas temporales o estacionales).
ALTER TABLE logro
    ADD COLUMN IF NOT EXISTS fecha_inicio DATE NULL,
    ADD COLUMN IF NOT EXISTS fecha_fin    DATE NULL;

-- Control: la fecha de fin debe ser posterior a la fecha de inicio.
ALTER TABLE logro
    ADD CONSTRAINT ck_logro_fechas
        CHECK (fecha_fin IS NULL OR fecha_inicio IS NULL OR fecha_fin > fecha_inicio);


-- 1.3.7
-- Agregar 1 campo entero y colocar un control para que no sea negativo
--

-- Se agrega el campo 'total_otorgados' para registrar cuántas veces
-- ha sido entregado este logro en la plataforma.
ALTER TABLE logro
    ADD COLUMN IF NOT EXISTS total_otorgados INT NOT NULL DEFAULT 0;

ALTER TABLE logro
    ADD CONSTRAINT ck_logro_total_otorgados
        CHECK (total_otorgados >= 0);


-- 1.3.8
-- Modificar el tamaño de un campo texto de la tabla renombrada
--

-- Se amplía 'categoria' de VARCHAR(80) a VARCHAR(120) para permitir
-- categorías con nombres más descriptivos.
ALTER TABLE logro
    ALTER COLUMN categoria TYPE VARCHAR(120);


-- 1.3.7 (bis)
-- Modificar el campo numérico y colocar un control de rango
--

-- Se modifica 'puntos' para asegurarse de que su valor esté en el rango
-- permitido de 0 a 1000 puntos por insignia.
ALTER TABLE logro
    ADD CONSTRAINT ck_logro_puntos
        CHECK (puntos IS NULL OR puntos BETWEEN 0 AND 1000);


-- 1.3.8 (bis)
-- Agregar un índice a la tabla (cualquier campo)
--

-- Se crea un índice sobre el campo 'categoria' para acelerar consultas
-- que filtren logros por tipo de categoría.
CREATE INDEX IF NOT EXISTS idx_logro_categoria
    ON logro(categoria);


--
-- 1.3.9
-- Eliminar una de las fechas
--

-- Se elimina 'fecha_fin' ya que se decide que los logros una vez
-- habilitados no caducan; solo tienen fecha de activación.
ALTER TABLE logro
    DROP COLUMN IF EXISTS fecha_fin;


-- 1.3.10
-- Borrar todos los datos de una tabla sin dejar traza
--

-- TRUNCATE elimina todos los registros de la tabla logro de forma
-- inmediata, sin registrar cada eliminación individual en el log,
-- lo que lo hace más eficiente que DELETE para limpiar tablas completas.
-- RESTART IDENTITY reinicia los contadores SERIAL.
-- CASCADE propaga el truncado a tablas dependientes.
TRUNCATE TABLE logro RESTART IDENTITY CASCADE;
